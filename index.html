<!-- JavaScript Application Logic -->
<script>
// Audio Control State
const audio = document.getElementById('bgMusic');
const audioToggle = document.getElementById('audioToggle');
const audioIcon = document.getElementById('audioIcon');
const audioText = document.getElementById('audioText');
let isPlaying = false;

audioToggle.addEventListener('click', () => {
if (isPlaying) {
audio.pause();
audioIcon.textContent = '🔇';
audioText.textContent = 'Soundtrack: Off';
isPlaying = false;
} else {
audio.play().catch(e => console.log("Audio play blocked by browser policy"));
audioIcon.textContent = '🎵';
audioText.textContent = 'Soundtrack: On';
isPlaying = true;
}
});

// Unlock Experience & Trigger Initial Confetti
function unlockExperience() {
// Attempt autoplay on interaction
audio.play().then(() => {
isPlaying = true;
audioIcon.textContent = '🎵';
audioText.textContent = 'Soundtrack: On';
}).catch(e => {});

// Trigger grand confetti entry
confetti({
particleCount: 100,
spread: 70,
origin: { y: 0.6 }
});

// Transition Sections
document.getElementById('sec-entry').classList.remove('active');
setTimeout(() => {
document.getElementById('sec-entry').style.display = 'none';
const dashboard = document.getElementById('sec-dashboard');
dashboard.classList.add('active');
}, 300);
}

// Tab Switching Logic
function switchTab(tabName) {
// Hide all contents
document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
// Reset all buttons
document.querySelectorAll('.tab-btn').forEach(btn => {
btn.className = "tab-btn px-5 py-2.5 rounded-full text-sm font-medium transition-all glass-panel text-zinc-400 hover:text-white";
});

// Show selected content
document.getElementById(`content-${tabName}`).classList.remove('hidden');
// Highlight selected button
const activeBtn = document.getElementById(`tab-${tabName}`);
activeBtn.className = "tab-btn px-5 py-2.5 rounded-full text-sm font-medium transition-all bg-pink-500 text-white shadow-lg shadow-pink-500/20";
}

// Interactive Sparkle Effect on Cards
function sparkleCard(element) {
confetti({
particleCount: 25,
spread: 50,
origin: {
x: (element.getBoundingClientRect().left + element.getBoundingClientRect().right) / 2 / window.innerWidth,
y: element.getBoundingClientRect().top / window.innerHeight
}
});
}

// Vault Surprise Button
function launchSurprise() {
const wishInput = document.getElementById('wishInput').value;

// Full screen confetti burst
var duration = 2.5 * 1000;
var end = Date.now() + duration;

(function frame() {
confetti({
particleCount: 4,
angle: 60,
spread: 55,
origin: { x: 0 }
});
confetti({
particleCount: 4,
angle: 120,
spread: 55,
origin: { x: 1 }
});

if (Date.now() < end) {
requestAnimationFrame(frame);
}
}());

if(wishInput.trim() !== "") {
alert(`Your wish: "${wishInput}" has been sent straight into the universe! Happy Birthday! ✨`);
document.getElementById('wishInput').value = "";
}
}
</script>
